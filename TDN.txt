// ==================== DIALOG ====================
function say(name, text, auto = false) {
G.dialogQueue.push({name, text, auto});
if(G.state !== 'dialog') {
G.state = 'dialog';
G.dialogIndex = 0;
nextDialog();
}
}

function nextDialog() {
if(G.dialogIndex >= G.dialogQueue.length) {
G.dialogQueue = []; G.dialogIndex = 0;
G.state = G.chaseActive ? 'chase' : 'gameplay';
document.getElementById('dialog-box').style.display = 'none';
return;
}
const d = G.dialogQueue[G.dialogIndex];
G.dialogName = d.name; G.dialogTarget = d.text;
G.dialogText = ''; G.dialogChar = 0; G.dialogTyping = true;
document.getElementById('dialog-name').textContent = d.name;
document.getElementById('dialog-box').style.display = 'block';
G.dialogIndex++;
}

function updateDialog(dt) {
if(!G.dialogTyping) {
if(keysPressed.x || keysPressed.sq) { keysPressed.x = false; keysPressed.sq = false; nextDialog(); }
return;
}
G.dialogChar += dt * 35;
if(G.dialogChar >= G.dialogTarget.length) {
G.dialogText = G.dialogTarget; G.dialogTyping = false;
} else {
G.dialogText = G.dialogTarget.slice(0, Math.floor(G.dialogChar));
}
document.getElementById('dialog-text').textContent = G.dialogText;
if(keysPressed.x || keysPressed.sq) {
keysPressed.x = false; keysPressed.sq = false;
G.dialogText = G.dialogTarget; G.dialogTyping = false;
}
}

document.getElementById('day-indicator').textContent = 'ДЕНЬ ' + G.day;
}

// ==================== INTERACTIONS ====================
function interact() {
const room = ROOMS[G.room];
let target = null, bestDist = 999;
for(let o of room.objects) {
if(!o.interact) continue;
const cx = o.x + o.w/2, cy = o.y + o.h/2;
const d = Math.hypot(cx - G.player.x, cy - G.player.y);
if(d < 40 && d < bestDist) { bestDist = d; target = o; }
}
if(!target) return;
const id = target.id;

if(id === 'tv') {
if(G.day <= 2) { say('','Телевизор показывает только белый шум... Или там кто-то есть?'); G.flicker = 0.5; }
else if(G.day <= 6) { say('','На экране — твое отражение. Оно моргает с задержкой на секунду.'); G.distort = 1; }
else say('','Телевизор показывает твою комнату. Сзади стоит ТО, что раньше было Димой.');
}

function tryHide() {
if(!G.chaseActive) return false;
const room = ROOMS[G.room];
for(let h of room.hideSpots) {
const cx = h.x + h.w/2, cy = h.y + h.h/2;
if(Math.hypot(cx - G.player.x, cy - G.player.y) < 50) {
G.player.hidden = true; G.hideTimer = 3 + rng() * 2;
say('','Ты спрятался... Сердце колотится. Шаги приближаются...');
return true;
}
}
return false;
}

// ==================== UPDATE ====================
function update(dt) {
if(G.state === 'title' || G.state === 'screamer') return;
if(G.state === 'dialog') { updateDialog(dt); return; }
if(G.state === 'transition') { G.flicker = Math.max(0, G.flicker - dt); return; }
if(G.state === 'ending') {
G.endingTimer += dt;
if(G.endingTimer > 4 && G.screamerPhase === 0) triggerScreamer();
return;
}

updateTime(dt);
updateHorror(dt);

let speed = G.chaseActive ? 45 : 30;
if(G.player.hidden) {
G.hideTimer -= dt;
if(G.hideTimer <= 0) { G.player.hidden = false; say('','Кажется, опасность миновала... Пока что.'); }
return;
}

let mx = 0, my = 0;
if(keys.up) my -= 1; if(keys.down) my += 1;
if(keys.left) mx -= 1; if(keys.right) mx += 1;
if(mx !== 0 || my !== 0) {
const len = Math.hypot(mx, my); mx /= len; my /= len;
if(keys.cir && G.chaseActive) speed *= 1.5;
G.player.x += mx * speed * dt; G.player.y += my * speed * dt;
if(Math.abs(mx) > Math.abs(my)) G.player.dir = mx > 0 ? 'right' : 'left';
else G.player.dir = my > 0 ? 'down' : 'up';
G.player.frame += dt * 8;
} else G.player.frame = 0;

const room = ROOMS[G.room];
G.player.x = Math.max(20, Math.min(room.w - 20, G.player.x));
G.player.y = Math.max(40, Math.min(room.h - 20, G.player.y));

for(let d of room.doors) {
if(G.player.x > d.x && G.player.x < d.x + d.w && G.player.y > d.y && G.player.y < d.y + d.h) {
G.room = d.to;
const sp = SPAWN[d.to];
if(d.side === 'top') { G.player.x = d.x + d.w/2; G.player.y = ROOMS[d.to].h - 25; }
else if(d.side === 'bottom') { G.player.x = d.x + d.w/2; G.player.y = 25; }
else if(d.side === 'left') { G.player.x = ROOMS[d.to].w - 25; G.player.y = d.y + d.h/2; }
else if(d.side === 'right') { G.player.x = 25; G.player.y = d.y + d.h/2; }
G.flicker = 0.3; break;
}
}

if(keysPressed.x) { keysPressed.x = false; interact(); }
if(keysPressed.cir) { keysPressed.cir = false; if(G.chaseActive) tryHide(); }

if(G.day >= 3 && G.day <= 6 && G.friends.length > 0) updateFriends(dt);
if(G.chaseActive) updateMonsters(dt);
updateEvents(dt);
}

function updateHorror(dt) {
G.flicker = Math.max(0, G.flicker - dt * 2);
G.shake = Math.max(0, G.shake - dt * 5);
G.distort = Math.max(0, G.distort - dt * 0.5);
if(G.day >= 2 && rng() < 0.002 * G.day) G.flicker = 0.3 + rng() * 0.5;
if(G.day >= 3 && rng() < 0.001 * G.day) G.shake = 0.5 + rng();
if(G.day >= 4 && rng() < 0.001) G.distort = 1;
}

function updateFriends(dt) {
for(let f of G.friends) {
f.timer = (f.timer || 0) + dt;
if(f.timer > 3 + rng() * 4) { f.timer = 0; f.tx = 40 + rng() * 240; f.ty = 50 + rng() * 100; }
f.x += (f.tx - f.x) * dt * 0.5; f.y += (f.ty - f.y) * dt * 0.5;
}
}

function updateMonsters(dt) {
for(let m of G.monsters) {
if(m.room === G.room && !G.player.hidden) {
const dx = G.player.x - m.x, dy = G.player.y - m.y;
const dist = Math.hypot(dx, dy);
if(dist > 5) { m.x += (dx/dist) * m.speed * 40 * dt; m.y += (dy/dist) * m.speed * 40 * dt; }
if(dist < 20) {
say('','Тебя поймали... Но все начинается сначала. Полночь еще не наступила.');
G.player.x = SPAWN[G.room].x; G.player.y = SPAWN[G.room].y;
G.player.hidden = true; G.hideTimer = 2;
}
} else {
m.timer += dt;
if(m.timer > 5 + rng() * 5) {
m.timer = 0;
if(rng() < 0.3) { m.room = G.room; m.x = G.player.x + (rng() - 0.5) * 100; m.y = G.player.y + (rng() - 0.5) * 100; }
}
}
}
}

function updateEvents(dt) {
if(G.day <= 2 && !G.eventFlags.footsteps && G.hour >= 21 && rng() < 0.01) {
G.eventFlags.footsteps = true;
say('','Ты слышишь четкие шаги в коридоре. Тяжелые. Медленные. Кто-то ходит.'); G.shake = 1;
}
if(G.day === 2 && !G.eventFlags.moved && G.hour >= 19) {
G.eventFlags.moved = true; say('','Стул стоит у двери. Ты не ставил его там.');
}
if(G.day === 3 && !G.eventFlags.friendsNotice && G.hour >= 22) {
G.eventFlags.friendsNotice = true;
say('Кирилл', 'Ребят... я слышал, как кто-то шептал из вентиляции. Это не ветер.');
say('Артем', '...Может, нам правда стоит уйти?');
say('Дима', 'Да бросьте. Мы же вместе. Ничего не случится.');
}
if(G.day === 4 && !G.eventFlags.distort && G.hour >= 20) {
G.eventFlags.distort = true; G.distort = 3;
say('','На мгновение лицо Артема исказилось. Глаза слишком широко. Рот слишком узко. Потом все стало нормально.');
}
if(G.day === 5 && !G.eventFlags.suffocate && G.hour >= 19) {
G.eventFlags.suffocate = true;
say('','Воздух стал густым, как вода. Дышать трудно. Друзья сидят неподвижно. Смотрят в одну точку.');
}
if(G.day === 6 && !G.eventFlags.reality && G.hour >= 21) {
G.eventFlags.reality = true;
say('','Стены дышат. Обои шевелятся. Дима смеется, но звук идет с задержкой. Артем моргает в такт твоему сердцебиению.');
G.flicker = 2;
}
}

function startEnding() {
G.state = 'ending'; G.chaseActive = false; G.endingTimer = 0;
say('','Ты просыпаешься в своей кровати. Мягкий свет утреннего солнца. Все было сном. Кошмарным, но всего лишь сном...');
setTimeout(() => { say('','Ты выдыхаешь с облегчением. Сердце успокаивается. Все хорошо. Все...'); }, 3000);
}

function triggerScreamer() {
G.screamerPhase = 1; G.state = 'screamer';
document.getElementById('screamer').style.display = 'flex';
drawScaryFace();
let flash = 0;
const flashInt = setInterval(() => {
flash++;
document.getElementById('screamer').style.background = flash % 2 === 0 ? '#000' : '#300';
if(flash > 10) clearInterval(flashInt);
}, 100);
}

// ==================== DRAWING ====================
function draw() {
ctx.fillStyle = '#000'; ctx.fillRect(0, 0, W, H);
if(G.state === 'title') return;
if(G.state === 'screamer') return;

let sx = 0, sy = 0;
if(G.shake > 0) { sx = (rng() - 0.5) * G.shake * 6; sy = (rng() - 0.5) * G.shake * 6; }
ctx.save(); ctx.translate(sx, sy);
if(G.distort > 0) ctx.translate((rng()-0.5)G.distort3, (rng()-0.5)G.distort3);

const room = ROOMS[G.room];
let camX = G.player.x - W/2, camY = G.player.y - H/2;
camX = Math.max(0, Math.min(room.w - W, camX));
camY = Math.max(0, Math.min(room.h - H, camY));
ctx.save(); ctx.translate(-camX, -camY);

ctx.fillStyle = room.floor; ctx.fillRect(0, 0, room.w, room.h);
ctx.fillStyle = 'rgba(0,0,0,0.05)';
for(let tx = 0; tx < room.w; tx += 20) {
for(let ty = 0; ty < room.h; ty += 20) {
if((tx + ty) % 40 === 0) ctx.fillRect(tx, ty, 20, 20);
}
}

ctx.fillStyle = room.color;
for(let w of room.walls) {
ctx.fillRect(w.x, w.y, w.w, w.h);
ctx.fillStyle = 'rgba(0,0,0,0.2)'; ctx.fillRect(w.x, w.y + w.h - 3, w.w, 3); ctx.fillStyle = room.color;
}

for(let d of room.doors) {
ctx.fillStyle = '#1a120a'; ctx.fillRect(d.x, d.y, d.w, d.h);
ctx.strokeStyle = '#333'; ctx.lineWidth = 1; ctx.strokeRect(d.x, d.y, d.w, d.h);
}

for(let o of room.objects) {
ctx.fillStyle = o.color; ctx.fillRect(o.x, o.y, o.w, o.h);
ctx.strokeStyle = 'rgba(0,0,0,0.3)'; ctx.lineWidth = 1; ctx.strokeRect(o.x, o.y, o.w, o.h);
if(o.interact) {
const cx = o.x + o.w/2, cy = o.y + o.h/2;
const dist = Math.hypot(cx - G.player.x, cy - G.player.y);
if(dist < 40) {
ctx.strokeStyle = 'rgba(255,255,200,0.4)'; ctx.strokeRect(o.x-1, o.y-1, o.w+2, o.h+2);
G.hint = '✕ ' + o.name; G.hintTimer = 0.5;
}
}
}

if(G.day >= 3 && G.day <= 6) {
for(let f of G.friends) {
drawCharacter(f.x, f.y, f.shirt, f.hair, f.face, G.distort > 1);
ctx.fillStyle = 'rgba(0,0,0,0.5)'; ctx.fillRect(f.x - 15, f.y - 22, 30, 8);
ctx.fillStyle = '#aaa'; ctx.font = '6px monospace'; ctx.textAlign = 'center'; ctx.fillText(f.name, f.x, f.y - 16);
}
}

if(G.chaseActive) {
for(let m of G.monsters) {
if(m.room === G.room) drawMonster(m.x, m.y, m.color);
}
}

if(!G.player.hidden) drawCharacter(G.player.x, G.player.y, '#48a', '#321', '#cba', false, true);

ctx.restore(); // camera
ctx.restore(); // shake

drawLighting(camX, camY);

if(G.flicker > 0) { ctx.fillStyle = rgba(0,0,0,${G.flicker * 0.7})`; ctx.fillRect(0, 0, W, H); }

const grad = ctx.createRadialGradient(W/2, H/2, 60, W/2, H/2, 140);
grad.addColorStop(0, 'rgba(0,0,0,0)');
grad.addColorStop(1, rgba(0,0,0,${0.5 + G.day * 0.05})`);
ctx.fillStyle = grad; ctx.fillRect(0, 0, W, H);

if(G.hintTimer > 0) {
G.hintTimer -= 0.016;
document.getElementById('hint-text').textContent = G.hint;
document.getElementById('hint-text').style.display = 'block';
} else document.getElementById('hint-text').style.display = 'none';
}

function drawCharacter(x, y, shirt, hair, face, distorted, isPlayer = false) {
const bob = Math.sin(G.player.frame) * 2;
const yb = y + bob;
ctx.fillStyle = 'rgba(0,0,0,0.3)'; ctx.fillRect(x - 6, y + 2, 12, 4);
ctx.fillStyle = shirt;
if(distorted && rng() < 0.3) ctx.fillStyle = '#511';
ctx.fillRect(x - 5, yb - 6, 10, 8);
ctx.fillStyle = face;
if(distorted && rng() < 0.3) ctx.fillStyle = '#ddd';
ctx.fillRect(x - 4, yb - 14, 8, 8);
ctx.fillStyle = hair; ctx.fillRect(x - 5, yb - 16, 10, 4);
if(distorted && rng() < 0.5) {
ctx.fillStyle = '#000'; ctx.fillRect(x - 3, yb - 12, 6, 3);
} else {
ctx.fillStyle = '#000'; ctx.fillRect(x - 3, yb - 12, 2, 2); ctx.fillRect(x + 1, yb - 12, 2, 2);
}
ctx.fillStyle = '#222'; ctx.fillRect(x - 4, yb + 2, 3, 5); ctx.fillRect(x + 1, yb + 2, 3, 5);
if(isPlayer) { ctx.fillStyle = 'rgba(255,255,255,0.3)'; ctx.fillRect(x - 1, yb - 18, 2, 2); }
}

function drawMonster(x, y, color) {
const twitch = Math.sin(Date.now() / 50) * 2;
ctx.fillStyle = color; ctx.fillRect(x - 8 + twitch, y - 10, 16, 12);
ctx.fillStyle = '#ddd'; ctx.fillRect(x - 10, y - 22, 20, 12);
ctx.fillStyle = '#000'; ctx.fillRect(x - 8, y - 18, 6, 6); ctx.fillRect(x + 2, y - 18, 6, 6);
ctx.fillStyle = '#f00'; ctx.fillRect(x - 7, y - 17, 2, 2); ctx.fillRect(x + 3, y - 17, 2, 2);
ctx.fillStyle = '#200'; ctx.fillRect(x - 6, y - 10, 12, 4);
ctx.fillStyle = '#fff';
for(let i = 0; i < 5; i++) ctx.fillRect(x - 5 + i * 3, y - 10, 1, 3);
ctx.fillStyle = color; ctx.fillRect(x - 14, y - 5, 8, 3); ctx.fillRect(x + 6, y - 2, 8, 3); ctx.fillRect(x - 3, y + 5, 3, 8);
}

function drawLighting(cx, cy) {
const px = G.player.x - cx, py = G.player.y - cy;
const grad = ctx.createRadialGradient(px, py, 20, px, py, 100 + G.day * 5);
grad.addColorStop(0, 'rgba(0,0,0,0)');
grad.addColorStop(0.5, rgba(0,0,0,${0.3 + G.day * 0.08})); grad.addColorStop(1,rgba(0,0,0,${0.7 + G.day * 0.05}));
ctx.fillStyle = grad; ctx.fillRect(0, 0, W, H);
}

function startGame() {
document.getElementById('title-screen').style.display = 'none';
G.state = 'gameplay';
setupDay(1);
gameRunning = true;
lastTime = performance.now();
requestAnimationFrame(loop);
}

function loop(time) {
if(!gameRunning) return;
const dt = Math.min((time - lastTime) / 1000, 0.1);
lastTime = time;
update(dt);
draw();
requestAnimationFrame(loop);
}
